export function familyGeometry(full,skeleton,scale){
  const wood=structuredClone(full.wood),leaves=structuredClone(full.leaves);
  if(!Number.isFinite(scale)||scale<=0)throw new Error('Invalid skeleton scale');
  const count=mesh=>mesh.position.length/3;
  for(const mesh of [wood,leaves]){
    const n=count(mesh);
    if(!Number.isInteger(n)||n<3||n>65535||mesh.normal.length!==n*3||mesh.uv.length!==n*2||mesh.index.length%3||mesh.index.length>100000*3)throw new Error('Invalid source geometry');
    for(const key of ['position','normal','uv'])if(mesh[key].some(x=>!Number.isFinite(x)))throw new Error('Nonfinite source geometry');
    if(mesh.index.some(x=>!Number.isInteger(x)||x<0||x>=n))throw new Error('Invalid source indices');
    mesh.wind=Array(n).fill(0);mesh.level=Array(n).fill(mesh===leaves?4:0);mesh.center=Array(n*3).fill(0);
  }
  wood.pivot=Array(count(wood)*3).fill(0);
  if(leaves.pivot.length!==leaves.position.length||leaves.pivot.some(x=>!Number.isFinite(x)))throw new Error('Invalid leaf pivots');
  // The pinned full mesh emits each branch's complete rings in skeleton order.
  let vertex=0;
  for(const branch of skeleton.branches){
    const segments=branch.segmentCount,sections=branch.sections;
    if(!Number.isInteger(segments)||segments<3||segments>64||sections.length<2)throw new Error('Invalid branch layout');
    const wrap=Math.max(1,Math.round(branch.baseRadius*scale*2*Math.PI*8))/8;
    let arc=0;
    for(let ring=0;ring<sections.length;ring++){
      if(ring)arc+=Math.hypot(...sections[ring].origin.map((x,i)=>(x-sections[ring-1].origin[i])*scale));
      for(let j=0;j<=segments;j++,vertex++){
        if(vertex>=count(wood))throw new Error('Branch layout exceeds wood buffer');
        wood.uv[vertex*2]=j/segments*wrap;wood.uv[vertex*2+1]=-arc;
      }
    }
  }
  if(vertex!==count(wood)||count(leaves)!==skeleton.leaves.length*8)throw new Error('Pinned skeleton mapping differs');
  for(let card=0;card<count(leaves)/4;card++){
    const first=card*4,leaf=Math.floor(card/2),tile=leaf%16;
    const p=leaves.position,axis=[0,1,2].map(k=>(p[first*3+k]+p[(first+3)*3+k]-p[(first+1)*3+k]-p[(first+2)*3+k])/2);
    const length=Math.hypot(...axis);if(Math.abs(length-1.2)>1e-4)throw new Error('Expected 1.2m source cards');
    // Register the drawn sprig base with its attachment (same .048m offset as
    // the existing Atlas family); local card dimensions remain unchanged.
    for(let j=0;j<4;j++)for(let k=0;k<3;k++)p[(first+j)*3+k]-=.048*axis[k]/length;
    const a=p.slice(first*3,first*3+3),b=p.slice((first+1)*3,(first+1)*3+3),c=p.slice((first+2)*3,(first+2)*3+3);
    const u=b.map((x,k)=>x-a[k]),v=c.map((x,k)=>x-a[k]),n=[u[1]*v[2]-u[2]*v[1],u[2]*v[0]-u[0]*v[2],u[0]*v[1]-u[1]*v[0]],norm=Math.hypot(...n);
    const center=[0,1,2].map(k=>(p[first*3+k]+p[(first+2)*3+k])/2);
    for(let j=0;j<4;j++){
      const i=first+j,x=leaves.uv[i*2],y=leaves.uv[i*2+1],inset=.5/2048,span=.25-2*inset;
      leaves.uv[i*2]=(tile%4)*.25+inset+x*span;
      leaves.uv[i*2+1]=Math.floor(tile/4)*.25+inset+(1-y)*span;
      leaves.wind[i]=Math.min(1,Math.max(0,leaves.pivot[i*3+1]/16))**2;
      for(let k=0;k<3;k++){leaves.normal[i*3+k]=n[k]/norm;leaves.center[i*3+k]=center[k];}
    }
  }
  // Mid detail keeps the first plane of every second leaf. Its card, pivot,
  // tile and wind are copied byte-for-byte from the near mesh.
  const mid={};const sizes={position:3,normal:3,uv:2,pivot:3,center:3,wind:1,level:1};
  for(const key of Object.keys(sizes))mid[key]=[];mid.index=[];
  for(let leaf=0;leaf<skeleton.leaves.length;leaf+=2){
    const base=mid.position.length/3;
    for(const [key,size] of Object.entries(sizes))mid[key].push(...leaves[key].slice(leaf*8*size,(leaf*8+4)*size));
    mid.index.push(base,base+1,base+2,base,base+2,base+3);
  }

type GLBDocument={json:any;binary:Uint8Array};
function readGLB(buffer:ArrayBuffer):GLBDocument {
  const view=new DataView(buffer);
  if(buffer.byteLength<28||view.getUint32(0,true)!==0x46546c67||view.getUint32(4,true)!==2||view.getUint32(8,true)!==buffer.byteLength)throw new Error('Expected a complete glTF 2 GLB');
  const length=view.getUint32(12,true),start=20+length;
  if(view.getUint32(16,true)!==0x4e4f534a||start+8>buffer.byteLength||view.getUint32(start+4,true)!==0x004e4942||start+8+view.getUint32(start,true)!==buffer.byteLength)throw new Error('Expected JSON and embedded binary chunks');
  const json=JSON.parse(new TextDecoder().decode(new Uint8Array(buffer,20,length)));
  if(json.buffers?.length!==1||json.buffers[0].uri||json.extensionsRequired?.length)throw new Error('Only self-contained author-6 GLBs are supported');
  if(json.nodes?.length!==1||json.nodes[0].mesh!==0||['matrix','translation','rotation','scale'].some(k=>json.nodes[0][k])||json.meshes?.length!==1||json.meshes[0].primitives?.length!==2)throw new Error('Expected one untransformed two-material tree');
  if(json.materials?.length!==2||json.materials[0].name!=='MAT_AtlasBark'||json.materials[1].name!=='MAT_AtlasFoliage'||json.images?.length!==4)throw new Error('Expected author-6 bark and foliage');
  if((json.materials[0].alphaMode??'OPAQUE')!=='OPAQUE'||json.materials[1].alphaMode!=='MASK'||(json.materials[1].alphaCutoff??.5)<=0)throw new Error('Tree foliage requires alpha MASK with a positive cutoff');

  for(const [index,p] of json.meshes[0].primitives.entries()){
    if(p.material!==index||p.mode!==undefined&&p.mode!==4||p.targets||Object.keys(p.attributes??{}).length!==8||!Object.keys(formats).every(k=>Number.isInteger(p.attributes?.[k])))throw new Error('Incomplete tree vertex contract');

function imageBytes(doc:GLBDocument,index:number){
  const image=doc.json.images[index],view=doc.json.bufferViews[image.bufferView];
  if(image.uri||image.mimeType!=='image/png'||!view||view.buffer!==0||!Number.isInteger(view.byteLength)||view.byteLength<=0)throw new Error('Expected an embedded PNG');

/** CPU reference for shader audits. Scale skeleton/attachment coordinates,
 * keeping every offset within a leaf card in metres, then rotate rigidly. */
export function treeVertexMeters(position:THREE.Vector3,pivot:THREE.Vector3,leaf:boolean,heightM:number,windWeight:number,phase:number,instancePhase=0){
  if(!Number.isFinite(heightM)||heightM<10||heightM>22)throw new Error('Prototype supports tree heights 10–22 m');
  if(!leaf)return position.clone().multiplyScalar(heightM/TREE_REFERENCE_HEIGHT_M);

}

const declarations=`

attribute vec3 _leaf_pivot;
attribute vec2 instanceTree;
uniform float atlasTreePhase;

  float angle=sin(atlasTreePhase+instanceTree.y)*.12*_wind;

function deform(material:THREE.Material,phase:{value:number}){

    shader.uniforms.atlasTreePhase=phase;

      if(_branch_level>3.5){objectNormal=atlasLeafRotation()*objectNormal;
        #ifdef USE_TANGENT
        objectTangent=atlasLeafRotation()*objectTangent;
        #endif
      }`);
    shader.vertexShader=shader.vertexShader.replace('#include <begin_vertex>',`#include <begin_vertex>
      transformed=_branch_level>3.5 ? _leaf_pivot*instanceTree.x+atlasLeafRotation()*(position-_leaf_pivot) : position*instanceTree.x;`);
  };
  material.customProgramCacheKey=()=> 'atlas-tree-physical-1';
}
