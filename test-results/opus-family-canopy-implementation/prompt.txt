function convexShader(name:string,geometry:THREE.BufferGeometry){

function percentile(values: number[], q: number): number {
  if (!values.length) return 0;
  values.sort((a, b) => a - b);
  return values[Math.max(0, Math.ceil(q * values.length) - 1)];

function maxNormAtAngle(px: number, py: number, pz: number, ox: number, oy: number, oz: number, s: number, angle: number): number {
  const c = Math.cos(angle), sn = Math.sin(angle);
  const x = px * s + ox * c - oy * sn;
  const y = py * s + ox * sn + oy * c;
  const z = pz * s + oz;
  return Math.hypot(x, y, z);
}
function maxDeformedVertexRadius(px: number, py: number, pz: number, qx: number, qy: number, qz: number,
  leaf: boolean, wind: number): number {
  if (!leaf) return Math.hypot(px, py, pz) * TREE_CANOPY_HEIGHT_RANGE_M[1] / TREE_CANOPY_REFERENCE_HEIGHT_M;
  const ox = px - qx, oy = py - qy, oz = pz - qz, maxAngle = TREE_CANOPY_WIND_MAX_ANGLE_RAD * wind;
  let maximum = 0;
  // For fixed angle, squared distance is convex in uniform skeleton scale, so
  // its maximum over 10..22 m is at an endpoint. Angular maxima occur at the
  // interval ends or where A cos(theta)+B sin(theta) has a stationary point.
  const A = qx * ox + qy * oy, B = -qx * oy + qy * ox;
  const candidates = [-maxAngle, maxAngle];
  const stationary = Math.atan2(B, A);
  for (let k = -2; k <= 2; k++) {
    const a = stationary + k * Math.PI;
    if (a >= -maxAngle && a <= maxAngle) candidates.push(a);
  }
  for (const height of TREE_CANOPY_HEIGHT_RANGE_M) {
    const s = height / TREE_CANOPY_REFERENCE_HEIGHT_M;
    for (const angle of candidates) maximum = Math.max(maximum, maxNormAtAngle(qx, qy, qz, ox, oy, oz, s, angle));
  }
  return maximum;
}

  // scale with the skeleton; card offsets remain metric and are added unchanged.

const srgbLinearLut = Float64Array.from({length: 256}, (_, byte) => {
  const v = byte / 255;
  return v <= .04045 ? v / 12.92 : ((v + .055) / 1.055) ** 2.4;
});

/** Alpha-weighted mean of straight-alpha sRGB RGBA byte pixels, returned in linear RGB. */
export function alphaWeightedLinearRgbMean(rgba: ArrayLike<number>): {linearRgb: [number, number, number]; alphaWeight: number; pixelCount: number} {
  if (!rgba || !Number.isSafeInteger(rgba.length) || rgba.length === 0 || rgba.length % 4 !== 0) throw new Error('Expected non-empty straight sRGB RGBA pixels');
  let weight = 0, r = 0, g = 0, b = 0;
  const byteArray = rgba instanceof Uint8Array || rgba instanceof Uint8ClampedArray;
  for (let i = 0; i < rgba.length; i += 4) {
    const red = rgba[i], green = rgba[i + 1], blue = rgba[i + 2], alpha = rgba[i + 3];
    if (!byteArray && (!Number.isInteger(red) || red < 0 || red > 255 || !Number.isInteger(green) || green < 0 || green > 255 ||
      !Number.isInteger(blue) || blue < 0 || blue > 255 || !Number.isInteger(alpha) || alpha < 0 || alpha > 255)) throw new Error('RGBA values must be integer bytes');
    const a = alpha / 255;
    weight += a; r += srgbLinearLut[red] * a; g += srgbLinearLut[green] * a; b += srgbLinearLut[blue] * a;
  }
  if (weight <= 0) throw new Error('RGBA pixels have no alpha weight');
  return {linearRgb: [r / weight, g / weight, b / weight], alphaWeight: weight, pixelCount: rgba.length / 4};
}

/** Texture-space color summary. `baseColorFactor` is already linear, matching
 * Three's material color; alpha cutoff coverage measures pixels, never canopy area. */
export function deriveTreeBaseColorStatistics(rgba: ArrayLike<number>,
  baseColorFactor: readonly [number, number, number, number], alphaCutoff: number) {
  if (!Array.isArray(baseColorFactor) || baseColorFactor.length !== 4 ||
    !baseColorFactor.every(x => Number.isFinite(x) && x >= 0 && x <= 1) ||
    !Number.isFinite(alphaCutoff) || alphaCutoff < 0 || alphaCutoff > 1) throw new Error('Invalid glTF base color factor or alpha cutoff');
  const mean = alphaWeightedLinearRgbMean(rgba);
  let covered = 0;
  for (let i = 3; i < rgba.length; i += 4) {
    const alpha = rgba[i];
    if (!(rgba instanceof Uint8Array || rgba instanceof Uint8ClampedArray) &&
      (!Number.isInteger(alpha) || alpha < 0 || alpha > 255)) throw new Error('RGBA values must be integer bytes');
    if (alpha / 255 * baseColorFactor[3] >= alphaCutoff) covered++;
  }
  return {
    linearRgb: [mean.linearRgb[0] * baseColorFactor[0], mean.linearRgb[1] * baseColorFactor[1], mean.linearRgb[2] * baseColorFactor[2]] as [number, number, number],
    baseCoverage: covered / mean.pixelCount,
    pixelCount: mean.pixelCount,
    alphaWeight: mean.alphaWeight,
  };
}

# Matching detailed trees and the distant canopy

Status: local implementation; visual acceptance and deployment pending.

The normal Oak/Pine package now supplies the same family information to nearby
mesh instances and the complete distant census. This follows the genuine Opus
[near/far review](opus-near-far-family-assessment-2026-09-28.md); it does not introduce another tree generator or species model.

## Loaded family data

Each loaded LOD produces its own four-band crown profile from its actual leaf
attachment pivots. The distant draw uses LOD1. Bands retain vertical extent,
horizontal visual radius and a separate metric leaf-card margin. Empty bands
do not draw. A separate conservative sphere includes both LOD vertex sets,
supported 10–22 m skeleton heights, leaf wind and the proposed band envelopes.
The visual percentile radius is never used alone as a culling bound.

Base-color images are sampled once at family load. Alpha-weighted sRGB samples
are converted to linear RGB and multiplied by the imported material factor.
This provides separate bark and foliage colors. Atlas alpha occupancy remains
texture metadata; it is not multiplied into the entire crown's opacity.

Family presentation is enabled for a validated offline foliage sidecar. Basic
trees and experimental families without the sidecar keep the legacy distant
representation. The existing normal Trees control remains opt-in.

## Shared dated selection

`PlanetForest` supplies its existing dated catalogue to `PlanetCanopy`.
Canonical cell ID, tree height, position, yaw and habitat membership do not
change. The shared eligibility helper selects only a loaded family supporting
the canonical height. Per-instance bark and foliage brightness use the same
recipe binding as the detailed instances.

Family and both brightness values occupy the previously unused third float
in the far parameter buffer. The packed integer is exact in Float32; brightness
quantization error is at most 0.2/255. Code zero selects the legacy shape and
color. There is no added per-tree GPU attribute.

## Shader and handoff

The existing one-quad-per-tree analytic renderer draws broadleaf ellipsoids or
stacked conifer cones from the loaded middle LOD profile. Skeleton coordinates
scale with canonical height; leaf-card margins remain metres. Derived colors
replace the unrelated legacy green and roll tint for supported families.
The existing actual-hit depth, normals, ownership and local atmosphere remain
in use. Family quads and close-fallback detection use conservative bounds.

A catalogue change or adopted asset-library revision starts a bounded parameter
retag: at most 2,048 retained slots and 2 ms per prepare call. Nearby handoffs
refresh their own tags immediately before pairing. Pairing requires the current
presentation version, matching packed appearance, canonical tree height and
existing ground/root agreement. Retagging preserves transforms and ownership.
The CPU version array uses four bytes per allocated far slot (917,504 bytes at
229,376 slots). Diagnostics report pending retags, processed slots and slice time.

## Evidence and remaining limits

The focused integration run currently passes 67 checks across the far renderer,
canopy lifecycle, detailed family attachment and retained forest work list.
It covers bounded parameter writes, untouched roots/rotations, dated retagging,
height and appearance mismatch rejection, and shared catalogue/library choice.
Shader-source composition checks are not proof of browser GPU compilation.
The metadata and runtime changes also pass the TypeScript check. Detailed-tree
diagnostics include separate handoff samples so browser checks can compare the
near canonical ID against its far slot's height, appearance code and ownership.

This is an analytic approximation. The trunk still uses the legacy trunk shape
with the family's bark color. Solid band envelopes can overfill a sparse Pine;
profile shape and linear albedo alone cannot reproduce leaf-card holes, normal
maps, transmission or shadows. No realism acceptance or performance improvement

ground cover, terrain and fauna remain part of the full implementation plan.

# Offline foliage mip delivery

Local implementation following the updated Opus review. CPU checks and an
actual-game GPU upload/readback check pass. The matched medium-distance images
show restored foliage; complete forest appearance and normal-control acceptance
remain open.

## Why this path

The current optional coverage mip builder does synchronous work during asset
loading: 276 ms in one local capture and 322 ms in the corrected lighting run.
The smallest levels also report zero alpha-test coverage. Baking beforehand
removes that generation work from travel, while a bounded chain avoids those
known empty levels. Neither change by itself proves believable distant foliage.

The existing assetctl KTX2 path always generates its own mips from one image and
converts GLB textures to `KHR_texture_basisu`. Atlas's custom tree parser currently
requires four PNG images and uses a GLTFLoader without a KTX2Loader. Converting
the complete asset format is deferred until the existing texture path is accepted.

## Delivery contract

- A named assetctl operation reads a validated brief and a source tree GLB.
  It selects the foliage material's base-colour map; bark and normal maps remain
  untouched. No caller-supplied executable or generator source is accepted.
- Bake straight RGBA lower levels using the existing alpha-weighted linear RGB
  filter and per-atlas-cell alpha-test coverage rule. The original base PNG
  remains embedded in the GLB.
- Emit lower-level pixel payloads plus a manifest recording source hashes,
  algorithm version, cutoff, atlas grid, exact dimensions, per-level coverage,
  output hashes, preserved license state and provenance. Use a fresh output
  directory and preserve the source.
- Trial endpoint: level 7 for a 2048-square, 4-by-4-cell atlas (four pixels per
  cell side). This is a visual trial setting, not an accepted universal policy.
  A full chain remains available as reference.
- Runtime validates the manifest against the source foliage bytes before
  installing all levels on the shared texture. Colour, depth and distance
  passes use the same map and alpha threshold. Invalid input rejects the staged
  family replacement and preserves the previously active forest library.

## Local runtime implementation

`TreeAssetOptions.foliageMipSidecar` accepts a manifest and an ArrayBuffer.
Each input to `loadForestTreeFamilies` can supply its own sidecar, since Oak and
Pine have different source PNGs. Choosing both the runtime builder and offline
sidecar is rejected. No sidecar is fetched implicitly by production yet.

The loader validates dimensions, atlas layout, cutoff, contiguous level layout,
coverage metadata and SHA-256 hashes of the embedded source PNG, full payload
and each level. It installs the original base image and straight-RGBA ImageData
lower levels before the first GPU upload. Both LODs and their shadow materials
share the same Texture. Failed validation leaves that texture untouched;
disposal releases the installed image sources along with the family.

The level-7 payload is 5,592,064 bytes per family. This removes pixel generation
from loading, but hashing, ImageData creation and GPU upload still cost time.
The uncompressed foliage chain alone occupies approximately 21.3 MiB on the
GPU; this excludes the other textures, geometry, shadow map and driver overhead.

The [focused CPU log](../test-results/offline-foliage-mips/cpu.log) records 34
passing checks across sidecar validation/lifecycle, mip generation, actual Oak
LOD parsing and family integration. These checks do not establish browser
upload correctness or improved frame time.

## Renderer compatibility

The installed Three WebGL2 regular-Texture upload path uses the supplied mip
count for `texStorage2D` when generated mips are disabled. Immutable storage
bounds the available mip range to its allocated levels; this supports a short
chain without a separate shader LOD clamp. See the
[WebGL2 specification](https://registry.khronos.org/webgl/specs/latest/2.0/) and
[OpenGL ES 3.0.6 texture storage/completeness rules](https://registry.khronos.org/OpenGL/specs/es/3.0/es_spec_3.0.pdf).
The subsequent [browser record](../test-results/offline-foliage-mips/README.md)
confirms two eight-level immutable foliage textures, complete level-7 readback
and endpoint coverage matching the baked payloads. This is one RTX 5090 fixture,
not acceptance across browsers/devices.

Validate the actual shared Texture upload, minimum-size sampling, original
colour/alpha, shadow agreement, close/middle movement and matching far handoff.
Measure payload, decode/upload time and memory as well as total cold loading.
Use the observed in-game date after navigation for all wind and replay checks.

## Actual-game result

At 390.58 m, a matched downward view submitted 250 Oak and 369 Pine middle-detail
trees in both modes, with the same date/pose and detailed lighting. The
[offline image](../test-results/offline-foliage-mips/mid-pair-down-390m-rerun/offline-mid-down-angle.png)
retains visibly fuller foliage than the
[default image](../test-results/offline-foliage-mips/mid-pair-down-390m-rerun/default-mid-down-angle.png).
It also exposes a pronounced mismatch with the larger, brighter legacy far
crowns. The foliage change is useful; it does not finish the near/far transition.

Sidecar setup measured 24.4 ms for Oak and 21.9 ms for Pine in the close fixture.
The complete pair still took roughly four seconds to load, and its sequential
default/offline frame profiles do not establish a speedup. Thirty-two sampled
tree bindings matched after returning to the observed date; browser and GL error
checks passed. These captures used committed runtime `2e00837`, before the
normal asset control and retained forest work list.
