function forestMaterial(vertexColors:boolean){
  const material=new THREE.MeshLambertMaterial({vertexColors,color:vertexColors?0xffffff:0x456c28});
  material.onBeforeCompile=shader=>{
    shader.vertexShader='attribute float instanceCoverage,instanceForestSeed;\nflat varying float vForestCoverage,vForestSeed;\n'+shader.vertexShader;
    shader.vertexShader=shader.vertexShader.replace('#include <begin_vertex>','#include <begin_vertex>\nvForestCoverage=instanceCoverage;vForestSeed=instanceForestSeed;');
    shader.fragmentShader='flat varying float vForestCoverage,vForestSeed;\n'+forestCoverageGLSL+shader.fragmentShader;
    shader.fragmentShader=shader.fragmentShader.replace('#include <alphahash_fragment>',`#ifdef USE_ALPHAHASH
      diffuseColor.a *= vForestCoverage;
      #include <alphahash_fragment>
      #else
      if(atlasForestHash(uint(vForestSeed+.5))>=atlasForestLevel(vForestCoverage))discard;
      #endif`);
  };
  material.customProgramCacheKey=()=>`forest-coverage-2/${vertexColors}`;return material;
}

function readGLB(buffer:ArrayBuffer):GLBDocument {
  const view=new DataView(buffer);
  if(buffer.byteLength<28||view.getUint32(0,true)!==0x46546c67||view.getUint32(4,true)!==2||view.getUint32(8,true)!==buffer.byteLength)throw new Error('Expected a complete glTF 2 GLB');
  const length=view.getUint32(12,true),start=20+length;
  if(view.getUint32(16,true)!==0x4e4f534a||start+8>buffer.byteLength||view.getUint32(start+4,true)!==0x004e4942||start+8+view.getUint32(start,true)!==buffer.byteLength)throw new Error('Expected JSON and embedded binary chunks');
  const json=JSON.parse(new TextDecoder().decode(new Uint8Array(buffer,20,length)));
  if(json.buffers?.length!==1||json.buffers[0].uri||json.extensionsRequired?.length)throw new Error('Only self-contained author-6 GLBs are supported');
  if(json.nodes?.length!==1||json.nodes[0].mesh!==0||['matrix','translation','rotation','scale'].some(k=>json.nodes[0][k])||json.meshes?.length!==1||json.meshes[0].primitives?.length!==2)throw new Error('Expected one untransformed two-material tree');
  if(json.materials?.length!==2||json.materials[0].name!=='MAT_AtlasBark'||json.materials[1].name!=='MAT_AtlasFoliage'||json.images?.length!==4)throw new Error('Expected author-6 bark and foliage');
  if((json.materials[0].alphaMode??'OPAQUE')!=='OPAQUE'||json.materials[1].alphaMode!=='MASK'||(json.materials[1].alphaCutoff??.5)<=0)throw new Error('Tree foliage requires alpha MASK with a positive cutoff');
  const formats:Record<string,string>={_WIND:'SCALAR',_BRANCH_LEVEL:'SCALAR',_LEAF_CENTER:'VEC3',_LEAF_PIVOT:'VEC3',POSITION:'VEC3',NORMAL:'VEC3',TEXCOORD_0:'VEC2',TANGENT:'VEC4'};
  for(const [index,p] of json.meshes[0].primitives.entries()){
    if(p.material!==index||p.mode!==undefined&&p.mode!==4||p.targets||Object.keys(p.attributes??{}).length!==8||!Object.keys(formats).every(k=>Number.isInteger(p.attributes?.[k])))throw new Error('Incomplete tree vertex contract');
    const count=json.accessors[p.attributes.POSITION]?.count;
    for(const [name,type] of Object.entries(formats)){
      const a=json.accessors[p.attributes[name]];
      if(!a||a.type!==type||a.componentType!==5126||a.normalized||a.sparse||!Number.isInteger(count)||count<1||a.count!==count)throw new Error(`Invalid tree accessor ${name}`);
    }
  }
  return {json,binary:new Uint8Array(buffer,start+8,view.getUint32(start,true))};
}
function imageBytes(doc:GLBDocument,index:number){
  const image=doc.json.images[index],view=doc.json.bufferViews[image.bufferView];
  if(image.uri||image.mimeType!=='image/png'||!view||view.buffer!==0||!Number.isInteger(view.byteLength)||view.byteLength<=0)throw new Error('Expected an embedded PNG');
  const start=view.byteOffset??0,end=start+view.byteLength;
  if(start<0||end>doc.binary.length)throw new Error('Image outside GLB binary');
  return doc.binary.subarray(start,end);
}

export function prepareTreeLODPair(lod0:ArrayBuffer,lod1:ArrayBuffer){
  const a=readGLB(lod0),b=readGLB(lod1);
  for(const key of ['materials','textures','samplers'])if(JSON.stringify(a.json[key])!==JSON.stringify(b.json[key]))throw new Error(`Tree LOD ${key} differ`);

    if(x.length!==y.length||x.some((v,k)=>v!==y[k]))throw new Error(`Tree LOD image ${i} differs`);
  }
  const json=structuredClone(b.json);
  json.materials=json.materials.map((m:any)=>({name:m.name}));delete json.images;delete json.textures;delete json.samplers;
  // Keep the original binary offsets. Unused compressed PNG bytes remain in
  // this temporary parse input, but do not allocate image/GPU resources.
  const text=new TextEncoder().encode(JSON.stringify(json)),length=(text.length+3)&~3;
  const buffer=new ArrayBuffer(28+length+b.binary.length),view=new DataView(buffer),bytes=new Uint8Array(buffer);
  view.setUint32(0,0x46546c67,true);view.setUint32(4,2,true);view.setUint32(8,buffer.byteLength,true);

}

/** CPU reference for shader audits. Scale skeleton/attachment coordinates,

export function treeVertexMeters(position:THREE.Vector3,pivot:THREE.Vector3,leaf:boolean,heightM:number,windWeight:number,phase:number,instancePhase=0){
  if(!Number.isFinite(heightM)||heightM<10||heightM>22)throw new Error('Prototype supports tree heights 10–22 m');
  if(!leaf)return position.clone().multiplyScalar(heightM/TREE_REFERENCE_HEIGHT_M);

}

const declarations=`
attribute float _wind, _branch_level;
attribute vec3 _leaf_pivot;
attribute vec2 instanceTree;
uniform float atlasTreePhase;
mat3 atlasLeafRotation(){
  float angle=sin(atlasTreePhase+instanceTree.y)*.12*_wind;
  float c=cos(angle),s=sin(angle);
  return mat3(c,s,0.,-s,c,0.,0.,0.,1.);
}
`;
function deform(material:THREE.Material,phase:{value:number}){
  material.onBeforeCompile=shader=>{
    shader.uniforms.atlasTreePhase=phase;
    shader.vertexShader=declarations+shader.vertexShader;
    shader.vertexShader=shader.vertexShader.replace('#include <beginnormal_vertex>',`#include <beginnormal_vertex>
      if(_branch_level>3.5){objectNormal=atlasLeafRotation()*objectNormal;
        #ifdef USE_TANGENT
        objectTangent=atlasLeafRotation()*objectTangent;
        #endif
      }`);
    shader.vertexShader=shader.vertexShader.replace('#include <begin_vertex>',`#include <begin_vertex>
      transformed=_branch_level>3.5 ? _leaf_pivot*instanceTree.x+atlasLeafRotation()*(position-_leaf_pivot) : position*instanceTree.x;`);
  };
  material.customProgramCacheKey=()=> 'atlas-tree-physical-1';
}

            if(newSlot&&forest&&(forest.getX(index)!==1||forest.getY(index)!==0)){

            }

           * transform/rebase updates deliberately retain its current colors. */

                attribute.setXYZ(index,...color);colorUpdates[part].include(index,3);colorUpdates[part].flush(attribute);
              }
            }
          },
          setCoverage(index:number,coverage:number,seed:number){

            if(forest.getX(index)!==value||forest.getY(index)!==seed){forest.setXY(index,value,seed);forestUpdates.include(index,2);forestUpdates.flush(forest);}
          },
          dispose(){if(batchDisposed)return;batchDisposed=true;activeBatches--;for(const mesh of meshes){mesh.geometry.dispose();mesh.customDepthMaterial?.dispose();mesh.customDistanceMaterial?.dispose();mesh.dispose();}group.clear();group.removeFromParent();},
        };
      },
      dispose(){if(disposed)return;if(activeBatches)throw new Error('Dispose tree batches before their family');disposed=true;dispose();},
    };
  }catch(error){dispose();throw error;}
}

/** Full coverage is 256, not 255: all 256 possible hash values must pass. */
export const quantizeForestCoverage=(coverage:number)=>Math.floor(Math.max(0,Math.min(1,coverage))*256+.5);
export const forestCoverageGLSL=`
uint atlasForestHash(uint seed){
  uvec2 pixel=uvec2(gl_FragCoord.xy);
  uint h=pixel.x*1664525u+pixel.y*1013904223u;
  h^=h>>16u;h*=2246822519u;h^=h>>13u;
  return (h+seed)&255u;
}
uint atlasForestLevel(float coverage){return uint(clamp(coverage,0.0,1.0)*256.0+.5);}
`;
/** CPU oracle for the exact unsigned integer shader operation. */
export function forestScreenHash(x:number,y:number,seed=0){
  let h=(Math.imul(x>>>0,1664525)+Math.imul(y>>>0,1013904223))>>>0;
  h=(h^(h>>>16))>>>0;h=Math.imul(h,2246822519)>>>0;h=(h^(h>>>13))>>>0;
  return (h+seed)&255;
}

export const TREE_RECIPE_VERSION='tree-recipe-1';
export const TREE_VARIANT_COUNT=8;
export const TREE_ASSET_VERSION='temperate-branching-v1';

export type TreeSpeciesRecipe=Readonly<{
  version:typeof TREE_RECIPE_VERSION; speciesId:string; lineageId:string;
  family:'branching'|'spire'; sourceFingerprint:string;
  heightM:number; crownWidthM:number; trunkDiameterM:number;

  leafLengthM:number; leafWidthM:number; leafDensity:number;

function hash(text:string){
  let h=2166136261;
  for(let i=0;i<text.length;i++){h^=text.charCodeAt(i);h=Math.imul(h,16777619);}
  h^=h>>>16;h=Math.imul(h,0x7feb352d);h^=h>>>15;h=Math.imul(h,0x846ca68b);
  return (h^(h>>>16))>>>0;
}
const unit=(...parts:unknown[])=>hash(JSON.stringify(parts))/0x100000000;

/** The caller supplies the already dated/eligible catalogue species. Rendezvous
 * assignment is independent of enumeration order and unrelated non-tree species.
 * Adding a tree species can transfer some assignments; it does not move cells. */
export function bindTreeSpecies(cellId:string,eligibleSpecies:readonly Species[]):Species|null {
  if(!cellId)throw new Error('Canonical cell id is required');
  let chosen:Species|null=null,best=-1;
  for(const species of eligibleSpecies){
    if(species.kind!=='tree'||(species.family!=='branching'&&species.family!=='spire'))continue;
    const score=hash(JSON.stringify([cellId,species.id,'tree-species-binding-1']));
    if(score>best||(score===best&&species.id<(chosen?.id??''))){chosen=species;best=score;}
  }
  return chosen;
}

/** Convert existing catalogue traits into a physical rendering envelope. The
 * trunk/crown relationship is an art heuristic, not a botanical simulation. */
export function treeSpeciesRecipe(species:Species):TreeSpeciesRecipe {
  if(!species.id||!species.lineageId)throw new Error('Catalogue species and lineage ids are required');
  if(species.kind!=='tree'||(species.family!=='branching'&&species.family!=='spire'))throw new Error('A catalogue tree species is required');
  const t=species.traits;
  for(const key of ['heightM','widthRatio','branchCount','branchAngle','curvature','leafLength','leafWidth','leafDensity'] as const)
    if(!Number.isFinite(t[key]))throw new Error(`Non-finite tree trait ${key}`);
  if(t.heightM<=0||t.widthRatio<=0||t.leafLength<=0||t.leafWidth<=0||t.branchCount<1||t.leafDensity<0||t.leafDensity>1)throw new Error('Invalid tree physical envelope');
  if(!/^#[0-9a-f]{6}$/i.test(t.color)||!/^#[0-9a-f]{6}$/i.test(t.woodColor))throw new Error('Tree colours must be sRGB hex');
  const source=[species.id,species.lineageId,species.family,t.heightM,t.widthRatio,t.branchCount,t.branchAngle,t.curvature,t.leafLength,t.leafWidth,t.leafDensity,t.color,t.woodColor];
  return Object.freeze({version:TREE_RECIPE_VERSION,speciesId:species.id,lineageId:species.lineageId,family:species.family,
    sourceFingerprint:hash(JSON.stringify(source)).toString(16).padStart(8,'0'),
    heightM:t.heightM,crownWidthM:t.heightM*t.widthRatio,
    trunkDiameterM:t.heightM*(species.family==='spire'?.026:.034),
    branchCount:Math.round(t.branchCount),branchAngleRad:t.branchAngle,curvature:t.curvature,
    leafLengthM:t.leafLength,leafWidthM:t.leafWidth,leafDensity:t.leafDensity,
    barkSRGB:t.woodColor,foliageSRGB:t.color});
}

export function treeStructuralVariant(recipe:TreeSpeciesRecipe,variant:number){
  if(!Number.isInteger(variant)||variant<0||variant>=TREE_VARIANT_COUNT)throw new Error('Tree variant is outside the bounded pool');
  const u=(channel:string)=>unit(recipe.version,recipe.speciesId,recipe.sourceFingerprint,variant,channel);
  const heightScale=.94+.12*u('height'),crownScale=.93+.14*u('crown');
  return Object.freeze({variant,branchSeed:hash(JSON.stringify([recipe.version,recipe.speciesId,variant,'branches'])),
    heightM:recipe.heightM*heightScale,crownWidthM:recipe.crownWidthM*heightScale*crownScale,
    trunkDiameterM:recipe.trunkDiameterM*heightScale,
    branchCount:Math.max(3,Math.round(recipe.branchCount*(.9+.2*u('branches')))),
    branchAngleRad:recipe.branchAngleRad+(.08*u('angle')-.04),
    curvature:recipe.curvature+(.08*u('curve')-.04),
    // Leaves keep their species dimensions when the whole tree grows taller.
    leafLengthM:recipe.leafLengthM,leafWidthM:recipe.leafWidthM,
    leafDensity:Math.min(1,recipe.leafDensity*(.94+.12*u('density')))});
}

/** Cheap per-instance shader parameters. Tints multiply separate bark/leaf

export function treeInstanceAppearance(recipe:TreeSpeciesRecipe,cellId:string){

  return Object.freeze({variant:Math.floor(u('variant')*TREE_VARIANT_COUNT),

    windPhase01:u('wind'),textureVariant:Math.floor(u('texture')*4)});
}

}

import * as THREE from 'three';
import {liveGlsl} from './tree-impostor-reference-shader.ts';

/** Experimental thin, two-sided leaf response. Uses the sampled base colour
 * times a linear tint for transmission: a bounded subset inspired by Khronos
 * diffuse transmission, not a KHR_materials_diffuse_transmission importer.
 * The host's specular/Fresnel calculations are retained verbatim. */
export type LeafTransmissionOptions={factor?:number;tint?:readonly [number,number,number]};
const VERSION='atlas-leaf-diffuse-1';
const installed=new WeakSet<THREE.Material>();

function replaceLiveOnce(source:string,anchor:string,replacement:string){
  const live=liveGlsl(source),at=live.indexOf(anchor);
  if(at<0||live.indexOf(anchor,at+anchor.length)>=0)throw new Error(`Leaf transmission shader hook changed: ${anchor}`);
  return source.slice(0,at)+replacement+source.slice(at+anchor.length);
}

const declarations=`
uniform float atlasLeafTransmission;
uniform vec3 atlasLeafTransmissionTint;
vec3 atlasLeafBackIrradiance = vec3(0.0);
vec3 atlasLeafBackIblIrradiance = vec3(0.0);
vec3 atlasLeafGeometricNormal = vec3(0.0);
`;

// Branch on a shared uniform so factor zero executes the exact original
// diffuse expression and does not sample opposite-hemisphere environment maps.
function split(original:string,expression:string){
  return `if (atlasLeafTransmission > 0.0) {\n${expression}\n} else {\n${original}\n}`;
}

/** Exported for source-contract tests. Does not mutate global ShaderChunk. */
export function patchLeafTransmissionShader(fragmentShader:string,physicalChunk=THREE.ShaderChunk.lights_physical_pars_fragment,normalChunk=THREE.ShaderChunk.normal_fragment_begin){
  if(liveGlsl(fragmentShader).includes('atlasLeafTransmission'))throw new Error('Leaf transmission is already installed');
  replaceLiveOnce(normalChunk,'vec3 nonPerturbedNormal = normal;','vec3 nonPerturbedNormal = normal;');
  let chunk=physicalChunk;
  const direct='reflectedLight.directDiffuse += irradiance * BRDF_Lambert( material.diffuseColor );';
  chunk=replaceLiveOnce(chunk,direct,split(direct,`
    vec3 backIrradiance = saturate(dot(-atlasLeafGeometricNormal, directLight.direction)) * directLight.color;
    reflectedLight.directDiffuse += ((1.0-atlasLeafTransmission)*irradiance
      + atlasLeafTransmission*atlasLeafTransmissionTint*backIrradiance) * BRDF_Lambert(material.diffuseColor);`));

  const area='reflectedLight.directDiffuse += lightColor * material.diffuseColor * LTC_Evaluate( normal, viewDir, position, mat3( 1.0 ), rectCoords );';
  chunk=replaceLiveOnce(chunk,area,split(area,`
    reflectedLight.directDiffuse += lightColor * material.diffuseColor * (
      (1.0-atlasLeafTransmission)*LTC_Evaluate(normal, viewDir, position, mat3(1.0), rectCoords)
      + atlasLeafTransmission*atlasLeafTransmissionTint*LTC_Evaluate(-atlasLeafGeometricNormal, viewDir, position, mat3(1.0), rectCoords));`));

  const indirect='reflectedLight.indirectDiffuse += irradiance * BRDF_Lambert( material.diffuseColor );';
  chunk=replaceLiveOnce(chunk,indirect,split(indirect,`
    reflectedLight.indirectDiffuse += ((1.0-atlasLeafTransmission)*irradiance
      + atlasLeafTransmission*atlasLeafTransmissionTint*atlasLeafBackIrradiance) * BRDF_Lambert(material.diffuseColor);`));

  const environment='reflectedLight.indirectDiffuse += diffuse * cosineWeightedIrradiance;';
  chunk=replaceLiveOnce(chunk,environment,split(environment,`
    reflectedLight.indirectDiffuse += diffuse * ((1.0-atlasLeafTransmission)*cosineWeightedIrradiance
      + atlasLeafTransmission*atlasLeafTransmissionTint*atlasLeafBackIblIrradiance*RECIPROCAL_PI);`));
  let shader=replaceLiveOnce(fragmentShader,'#include <lights_physical_pars_fragment>',declarations+chunk);
  shader=replaceLiveOnce(shader,'#include <lights_fragment_begin>',`
    // nonPerturbedNormal already includes vertex wind and DoubleSide flipping.
    // Transmission follows the thin surface, not bumps that can cross its side.
    atlasLeafGeometricNormal = nonPerturbedNormal;
    #include <lights_fragment_begin>
    if (atlasLeafTransmission > 0.0) {
      atlasLeafBackIrradiance = getAmbientLightIrradiance(ambientLightColor);
      #ifdef USE_LIGHT_PROBES
        atlasLeafBackIrradiance += getLightProbeIrradiance(lightProbe, -atlasLeafGeometricNormal);
      #endif
      #if NUM_HEMI_LIGHTS > 0
        #pragma unroll_loop_start
        for ( int i = 0; i < NUM_HEMI_LIGHTS; i ++ ) {
          atlasLeafBackIrradiance += getHemisphereLightIrradiance(hemisphereLights[ i ], -atlasLeafGeometricNormal);
        }
        #pragma unroll_loop_end
      #endif
    }`);
  shader=replaceLiveOnce(shader,'#include <lights_fragment_maps>',`#include <lights_fragment_maps>
    #if defined(USE_ENVMAP) && defined(STANDARD) && defined(ENVMAP_TYPE_CUBE_UV)
      if (atlasLeafTransmission > 0.0) {
        atlasLeafBackIblIrradiance = getIBLIrradiance(-atlasLeafGeometricNormal);
      }
    #endif`);
  // Both opposite irradiances must exist before the host calls its indirect REs.
  const live=liveGlsl(shader),end=live.indexOf('#include <lights_fragment_end>');
  if(end<0||live.indexOf('#include <lights_fragment_end>',end+1)>=0||end<live.lastIndexOf('atlasLeafBackIblIrradiance = getIBLIrradiance'))throw new Error('Leaf transmission indirect-light order changed');
  const normals=live.indexOf('#include <normal_fragment_maps>'),begin=live.indexOf('#include <lights_fragment_begin>'),maps=live.indexOf('#include <lights_fragment_maps>');
  if(normals<0||live.indexOf('#include <normal_fragment_maps>',normals+1)>=0||normals>begin||begin>maps||maps>end)throw new Error('Leaf transmission normal/light order changed');
  return shader;
}

function values(options:LeafTransmissionOptions){
  if(options===null||typeof options!=='object'||Array.isArray(options)||Object.keys(options).some(key=>key!=='factor'&&key!=='tint'))throw new Error('Leaf transmission options must contain only factor and tint');
  const factor=options.factor===undefined?0:options.factor,tint=options.tint===undefined?[1,1,1]:options.tint;
  if(typeof factor!=='number'||!Number.isFinite(factor)||factor<0||factor>1||!Array.isArray(tint)||tint.length!==3||[0,1,2].some(i=>!Number.isFinite(tint[i])||tint[i]<0||tint[i]>1))throw new Error('Leaf transmission factor and linear tint must be in [0,1]');
  return {factor,tint:[tint[0],tint[1],tint[2]] as const};
}

function validateMaterial(material:THREE.MeshStandardMaterial){
  const physical=material as THREE.MeshPhysicalMaterial;
  if(!material.isMeshStandardMaterial||material.side!==THREE.DoubleSide||material.metalness!==0||material.metalnessMap||material.lightMap||material.aoMap||(physical.transmission??0)!==0||physical.transmissionMap)throw new Error('Leaf transmission requires a two-sided dielectric without metalness maps, lightmaps, AO maps or glass transmission');
}

/** Call AFTER the wind/deformation adapter. Shared uniforms work for all
 * instances and LODs; depth/distance materials are deliberately not changed.
 * Material flags affecting the supported domain must remain valid at compile.
 * Reapply explicitly if a material is cloned (Three does not clone hooks). */
export function installLeafTransmission(material:THREE.MeshStandardMaterial,options:LeafTransmissionOptions={}){
  const initial=values(options);validateMaterial(material);
  if(installed.has(material))throw new Error('Leaf transmission is already installed on this material');
  const factor={value:initial.factor},tint={value:new THREE.Vector3(...initial.tint)};
  const previousCompile=material.onBeforeCompile;
  // Capture the default key before installing our wrapper: Three's default key
  // reads this.onBeforeCompile and would otherwise describe the wrapper itself.
  const previousKey=material.customProgramCacheKey.bind(material);
  const baseKey=previousKey();
  const defaultKey=material.customProgramCacheKey===THREE.Material.prototype.customProgramCacheKey;
  const compile:THREE.Material['onBeforeCompile']=function(this:THREE.Material,shader,renderer){
    validateMaterial(material);
    previousCompile.call(this,shader,renderer);
    const patched=patchLeafTransmissionShader(shader.fragmentShader);
    if(shader.uniforms.atlasLeafTransmission||shader.uniforms.atlasLeafTransmissionTint)throw new Error('Leaf transmission uniform collision');
    shader.uniforms.atlasLeafTransmission=factor;
    shader.uniforms.atlasLeafTransmissionTint=tint;
    shader.fragmentShader=patched;
  };
  const key=()=>`${defaultKey?baseKey:previousKey()}|${VERSION}`;
  material.onBeforeCompile=compile;material.customProgramCacheKey=key;
  installed.add(material);material.needsUpdate=true;
  return {
    assertInstalled(){if(material.onBeforeCompile!==compile||material.customProgramCacheKey!==key)throw new Error('Leaf transmission hook was replaced after installation');},
    /** Values update atomically; no material/program per tree or per setting. */
    set(options:LeafTransmissionOptions){const next=values(options);factor.value=next.factor;tint.value.set(...next.tint);},
    get factor(){return factor.value;},
    get tint():readonly [number,number,number]{return [tint.value.x,tint.value.y,tint.value.z];},
  };
}

import * as THREE from 'three';

/** Shared approximation for local solid surfaces. This preserves the existing
 * terrain response; it is not a multiple-scattering atmosphere model. */
export const LOCAL_AERIAL_VERSION='local-aerial-1';
const attachedMaterials=new WeakSet<THREE.Material>();
export const localAerialGLSL=`
vec3 atlasApplyLocalAir(vec3 radiance,float distanceM,float eyeHeightM,float surfaceHeightM,float sunCosine,float enabled){
  float airHeight=max(0.0,(eyeHeightM+surfaceHeightM)*.5);
  float airFog=enabled*min(.92,1.0-exp(-distanceM/65000.0*exp(-airHeight/14000.0)));
  float daylight=smoothstep(-.15,.15,sunCosine);
  radiance=mix(radiance,mix(vec3(.020,.032,.055),vec3(.23,.43,.64),daylight),airFog);
  return mix(radiance,max(radiance,vec3(.007,.011,.018)),airFog);
}
`;

export function localAerialReference(radiance:readonly number[],distanceM:number,eyeHeightM:number,surfaceHeightM:number,sunCosine:number,enabled=1){
  const height=Math.max(0,(eyeHeightM+surfaceHeightM)*.5);
  const fog=enabled*Math.min(.92,1-Math.exp(-Math.max(0,distanceM)/65000*Math.exp(-height/14000)));
  const t=Math.max(0,Math.min(1,(sunCosine+.15)/.3)),day=t*t*(3-2*t);
  const night=[.020,.032,.055],noon=[.23,.43,.64],floor=[.007,.011,.018];
  return radiance.map((v,i)=>{const haze=night[i]+(noon[i]-night[i])*day,x=v+(haze-v)*fog;return x+(Math.max(x,floor[i])-x)*fog;});
}

/** Instance transforms remain in their existing local precision frames. The
 * GPU uses view-space distance; the planet-relative position is needed only
 * for kilometre-scale density and daylight, not fine terrain or collision.
 * Each supplied mesh must own its material (one material per batch is enough).
 * An explicit frame permits identity-transformed batches in that SAME frame to
 * share this material and its render callback. It is not per-object instancing. */
export function attachLocalAerial(mesh:THREE.Mesh,planet:THREE.Object3D,radiusM:number,sun:{value:THREE.Vector3},enabled={value:1},eyeHeight?:{value:number},surfaceHit?:{local:string;view:string},frame:THREE.Object3D=mesh){
  if(Array.isArray(mesh.material))throw new Error('Local aerial attachment expects one material per batch');
  const material=mesh.material;
  if(attachedMaterials.has(material))throw new Error('Local aerial material is already attached');
  attachedMaterials.add(material);
  const uniforms={
    atlasAirObjectToPlanet:{value:new THREE.Matrix4()},atlasAirRadius:{value:radiusM},
    atlasAirMetresPerViewUnit:{value:1},atlasAirEyeHeight:eyeHeight??{value:0},atlasAirSun:sun,
    atlasAirEnabled:enabled,
  };
  const compile=material.onBeforeCompile;
  const inheritedKey=material.customProgramCacheKey===THREE.Material.prototype.customProgramCacheKey?()=>compile.toString():material.customProgramCacheKey.bind(material);
  material.onBeforeCompile=(shader,renderer)=>{
    compile.call(material,shader,renderer);
    const missing=!shader.vertexShader.includes('#include <project_vertex>')?'project_vertex':!shader.fragmentShader.includes('#include <opaque_fragment>')?'opaque_fragment':null;
    if(missing){material.userData.localAerialVersion=`failed:${missing}`;return;}
    material.userData.localAerialVersion=LOCAL_AERIAL_VERSION;
    Object.assign(shader.uniforms,uniforms);
    shader.vertexShader='uniform mat4 atlasAirObjectToPlanet; varying vec3 atlasAirPlanetPosition,atlasAirViewPosition;\n'+shader.vertexShader;
    shader.vertexShader=shader.vertexShader.replace('#include <project_vertex>',`#include <project_vertex>
      vec4 atlasAirLocalPosition=vec4(transformed,1.0);
      #ifdef USE_INSTANCING
        atlasAirLocalPosition=instanceMatrix*atlasAirLocalPosition;
      #endif
      atlasAirPlanetPosition=(atlasAirObjectToPlanet*atlasAirLocalPosition).xyz;
      atlasAirViewPosition=mvPosition.xyz;`);
    shader.fragmentShader=`uniform mat4 atlasAirObjectToPlanet;
      uniform float atlasAirRadius,atlasAirMetresPerViewUnit,atlasAirEyeHeight,atlasAirEnabled;
      uniform vec3 atlasAirSun;varying vec3 atlasAirPlanetPosition,atlasAirViewPosition;
      ${localAerialGLSL}\n`+shader.fragmentShader;
    shader.fragmentShader=shader.fragmentShader.replace('#include <opaque_fragment>',`
      vec3 atlasSurfacePlanet=${surfaceHit?`(atlasAirObjectToPlanet*vec4(${surfaceHit.local},1.0)).xyz`:'atlasAirPlanetPosition'};
      outgoingLight=atlasApplyLocalAir(outgoingLight,length(${surfaceHit?.view??'atlasAirViewPosition'})*atlasAirMetresPerViewUnit,
        atlasAirEyeHeight,(length(atlasSurfacePlanet)-1.0)*atlasAirRadius,
        dot(normalize(atlasSurfacePlanet),atlasAirSun),atlasAirEnabled);
      #include <opaque_fragment>`);
  };
  material.customProgramCacheKey=()=>inheritedKey()+'/'+LOCAL_AERIAL_VERSION+'/'+(surfaceHit?`${surfaceHit.local}/${surfaceHit.view}`:'mesh');
  material.userData.localAerialVersion=LOCAL_AERIAL_VERSION;material.needsUpdate=true;
  const inversePlanet=new THREE.Matrix4(),eye=new THREE.Vector3(),scale=new THREE.Vector3();
  const before=mesh.onBeforeRender;
  mesh.onBeforeRender=(renderer,scene,camera,geometry,mat,group)=>{
    before.call(mesh,renderer,scene,camera,geometry,mat,group);
    inversePlanet.copy(planet.matrixWorld).invert();
    uniforms.atlasAirObjectToPlanet.value.multiplyMatrices(inversePlanet,frame.matrixWorld);
    eye.setFromMatrixPosition(camera.matrixWorld).applyMatrix4(inversePlanet);
    if(!eyeHeight)uniforms.atlasAirEyeHeight.value=(eye.length()-1)*radiusM;
    scale.setFromMatrixScale(planet.matrixWorld);
    uniforms.atlasAirMetresPerViewUnit.value=radiusM/Math.max(1e-20,Math.abs(scale.x));
  };
  return uniforms;
}

import type {BufferAttribute} from 'three';

/** Component ranges, not byte ranges. Pending GPU uploads survive invisible
 * frames. Keep at most eight ranges by merging the smallest gaps. */
export class InstanceUpdateRanges {
  private ranges:{start:number;end:number}[]=[];
  include(index:number,components:number){
    this.merge(index*components,(index+1)*components);
  }
  private merge(start:number,end:number){
    for(let i=this.ranges.length-1;i>=0;i--){
      const range=this.ranges[i];if(range.end<start||range.start>end)continue;
      start=Math.min(start,range.start);end=Math.max(end,range.end);this.ranges.splice(i,1);
    }
    this.ranges.push({start,end});this.ranges.sort((a,b)=>a.start-b.start);
    while(this.ranges.length>8){
      let closest=0;
      for(let i=1;i<this.ranges.length-1;i++)if(this.ranges[i+1].start-this.ranges[i].end<this.ranges[closest+1].start-this.ranges[closest].end)closest=i;
      this.ranges[closest].end=this.ranges[closest+1].end;this.ranges.splice(closest+1,1);
    }
  }
  flush(attribute:BufferAttribute){
    if(!this.ranges.length)return 0;
    for(const range of attribute.updateRanges)this.merge(range.start,range.start+range.count);
    attribute.clearUpdateRanges();let components=0;
    for(const range of this.ranges){attribute.addUpdateRange(range.start,range.end-range.start);components+=range.end-range.start;}
    attribute.needsUpdate=true;this.ranges.length=0;
    return components*attribute.array.BYTES_PER_ELEMENT;
  }
}
